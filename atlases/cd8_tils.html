pts = {
  const text = await FileAttachment("../atlas-data/cd8_tils_umap_full.csv").text();
  // ★ NEWLINE VIA fromCharCode, NOT AN ESCAPE, AND THE COMMENT ITSELF PROVED IT. A backslash-n written in this
  // template is an ESCAPE IN THE PYTHON SOURCE: Python resolves it to a real newline when it
  // parses the string literal, so the .qmd received an actual line break inside a JS string.
  // That split the statement across lines, broke the cell, and made Quarto render the entire
  // block as literal text instead of executing it -- which looks like a Quarto echo problem
  // and is not one. No escape sequence means nothing left to collapse.
  const NL = String.fromCharCode(10);
  const body = text.slice(text.indexOf(NL) + 1);
  const rows = body.split(NL);
  const n = rows[rows.length - 1] === "" ? rows.length - 1 : rows.length;
  const x = new Float32Array(n), y = new Float32Array(n), s = new Uint8Array(n);
  for (let i = 0; i < n; i++) {
    const line = rows[i];
    const a = line.indexOf(","), b = line.indexOf(",", a + 1);
    x[i] = +line.slice(0, a); y[i] = +line.slice(a + 1, b); s[i] = +line.slice(b + 1);
  }
  return {x, y, s, n};
}

// ★ DRAW ORDER IS BY RARITY, COMMONEST FIRST. This is the whole reason the point view earns its
// place next to the binned one: the modal hex map hides rare states entirely (huARdb's Tex-Tpex is
// the most common state in ONE hexagon), whereas painting rare states LAST puts them on top of the
// dense ones instead of under them.
drawOrder = {
  const counts = new Map();
  for (let i = 0; i < pts.n; i++) counts.set(pts.s[i], (counts.get(pts.s[i]) || 0) + 1);
  return [...counts.entries()].sort((a, b) => b[1] - a[1]).map(d => d[0]);
}

// ★ EVERY CELL, DRAWN INDIVIDUALLY. No sampling and no binning: 10,045 points straight onto a
// 2D canvas. A device-pixel-ratio backing store keeps it sharp on retina displays without
// multiplying the work, since the cost here is fill calls rather than DOM.
// ★ NO TOOLTIP ON THIS VIEW, DELIBERATELY. Hit-testing a million points needs a quadtree, and the
// only per-cell attribute is state, which the legend already gives. The "sampled cells" view keeps
// Plot's tooltips for anyone who wants to interrogate individual points.
// ★ A FUNCTION, NOT A CELL HOLDING A DOM NODE. As a value cell this rendered an object-inspector
// dump instead of the map in the default view. `//| output: false` only adds class="hidden" to the
// wrapper; the runtime still attaches an Inspector to this cell and parents the canvas there
// FIRST. The consuming cell then finds value.parentNode !== its own node and falls back to
// inspect(value), printing an object-inspector dump. Returning a builder means each consumer gets a
// fresh node that nothing else has adopted.
canvasMap = function () {
  const W = 720, H = 440, PAD = 8, R = 0.8;
  const dpr = Math.min(window.devicePixelRatio || 1, 2);
  const wrap = html`<div style="display:flex;gap:12px;align-items:flex-start"></div>`;
  const cv = html`<canvas width=${W * dpr} height=${H * dpr}
                    style="width:${W}px;height:${H}px"></canvas>`;
  const ctx = cv.getContext("2d");
  ctx.scale(dpr, dpr);

  let x0 = Infinity, x1 = -Infinity, y0 = Infinity, y1 = -Infinity;
  for (let i = 0; i < pts.n; i++) {
    if (pts.x[i] < x0) x0 = pts.x[i]; if (pts.x[i] > x1) x1 = pts.x[i];
    if (pts.y[i] < y0) y0 = pts.y[i]; if (pts.y[i] > y1) y1 = pts.y[i];
  }
  const sx = (W - 2 * PAD) / (x1 - x0), sy = (H - 2 * PAD) / (y1 - y0);
  const k = Math.min(sx, sy);
  const ox = PAD + ((W - 2 * PAD) - (x1 - x0) * k) / 2;
  const oy = PAD + ((H - 2 * PAD) - (y1 - y0) * k) / 2;
  const colour = i => stateColours[i];   // NOT tableau10[i%10]: that wrapped 9 of 19 states

  ctx.globalAlpha = 0.55;
  for (const code of drawOrder) {            // commonest first, so rare states land on top
    ctx.fillStyle = colour(code);
    ctx.beginPath();
    for (let i = 0; i < pts.n; i++) {
      if (pts.s[i] !== code) continue;
      const px = ox + (pts.x[i] - x0) * k, py = H - (oy + (pts.y[i] - y0) * k);
      ctx.moveTo(px, py); ctx.arc(px, py, R, 0, 6.2832);
    }
    ctx.fill();
  }

  const legend = html`<div style="font-size:11px;line-height:1.5;max-width:180px">
    ${drawOrder.map(code => html`<div><span style="display:inline-block;width:.65em;height:.65em;
      background:${colour(code)};margin-right:.4em;border-radius:2px"></span>${stateNames[code]}</div>`)}
  </div>`;
  wrap.append(cv, legend);
  return wrap;
}

// ★ MEAN AND DETECTION RATE, NOT A VIOLIN, and computed on CELLS rather than on the hexagons.
// Single-cell expression is zero-inflated, so a violin is dominated by the zero spike and its
// median reads 0 for gene after gene; a summary over hexagons would additionally weight states by
// how many bins they occupy rather than how many cells they have. Dot area is the fraction of the
// state's cells with any detected transcript, which is what makes a low mean readable as "few
// cells express it" rather than "all express it weakly".
{
  const g = geneA;
  // ★ EVERY BRANCH SAYS SOMETHING. Returning md`` gave a TITLED, COMPLETELY BLANK card whenever
  // nothing was entered or the gene was outside the shared set, while the map card beside it
  // printed a message for the same input. A blank card reads as a broken page.
  if (!g) return md`Enter a gene to see how it is distributed across this atlas's states.`;
  const e = geneIdx.genes[g];
  if (!e) return md`**${g} is not on this map**, so there is no per-state summary. The note in the
    sidebar explains which references carry it.`;
  // ★ THE SAME ZERO-WIDTH DOMAIN TRAP AS THE MAP, AND IT WAS FIXED ONLY THERE. With p99 == 0 the
  // colour domain is [0, 0]; d3 maps every value to t=0.5, so an UNDETECTED gene drew a mid-ramp
  // teal dot and a legend reading 0.000000 at both ends.
  if (e.p99 <= 0) return md`**${g} is not detected in this atlas** (${
    (100 * e.detect).toFixed(2)}% of cells), so there is no distribution to summarise.`;
  const S = geneIdx.states, nS = S.length;
  const gi = geneIdx.gene_order.indexOf(g);
  if (gi < 0) return md`**${g} has no per-state summary in this atlas**, so there is nothing to
    show here.`;
  const rows = S.map((st, j) => ({
    state: st,
    mean: byState[(gi * nS + j) * 2] / 255 * e.p99,
    detect: byState[(gi * nS + j) * 2 + 1] / 255,
    n: geneIdx.state_n[j]
  }));
  return Plot.plot({
    height: 40 + 22 * nS, marginLeft: 190, width: 620,
    x: {label: `${g} mean lognorm`, grid: true},
    y: {label: null, domain: rows.slice().sort((a, b) => b.mean - a.mean).map(d => d.state)},
    // ★ range STARTS AT 0 so the area really is proportional to the detection fraction. With a
    // floor of 2 a gene detected in ZERO cells still drew a visible dot, and the area ratios ran
    // 0.05 / 0.22 / 0.60 / 1.00 against the 0 / 0.10 / 0.50 / 1.00 the caption claims.
    r: {range: [0, 9], domain: [0, 1]},
    color: {scheme: "YlGnBu", domain: [0, e.p99], legend: true, label: "mean"},
    marks: [
      Plot.ruleY(rows, {y: "state", x1: 0, x2: "mean", stroke: "#E4E7EA"}),
      Plot.dot(rows, {y: "state", x: "mean", r: "detect", fill: "mean", stroke: "#33404D",
                      strokeWidth: 0.5, tip: true,
                      channels: {"% detected": d => (100 * d.detect).toFixed(1) + "%",
                                 "cells in state": "n"}})
    ]
  });
}

The two gene maps use independent colour scales, so compare where each gene’s signal sits, not how dark the two look against each other. A housekeeping gene would otherwise flatten a cytokine. On the per-state card, dot area is the fraction of that state’s cells with any detected transcript. A gene can carry a low mean because few cells express it or because all express it weakly, and those are different claims; the two channels separate them.

// One bucket per lookup, after which every other gene in that bucket is served from the browser
// cache. See docs/rse/specs/plan-atlas-gene-lookup.md for why this
// beats one file per gene (56,154 files) and why HTTP Range was rejected (dead over file://).

// so it is deliberately NOT stated as a single number anywhere reader-facing.
geneVals = async function (g) {
  const e = geneIdx.genes[g];
  if (!e) return null;
  const buf = await (await fetch(
    `../atlas-data/genes/cd8_tils/b${String(e.b).padStart(3, "0")}.bin`)).arrayBuffer();
  const n = geneIdx.n_hex;
  const row = new Uint8Array(buf).subarray(e.i * n, (e.i + 1) * n);
  // uint8 was quantised over [0, p99] with clipping, so this inverts to expression units and the
  // top of the ramp is a 99th percentile, NOT the maximum. The legend says so.
  return {v: Array.from(row, x => x / 255 * e.p99), p99: e.p99, detect: e.detect,
          // ★ p99 == 0 MEANS THE GENE IS EFFECTIVELY UNDETECTED IN THIS ATLAS, and it must be
          // reported rather than drawn. A zero-width colour domain makes d3 map every hexagon to
          // t=0.5, so the map came out a uniform mid-ramp teal that reads as MODERATE EXPRESSION
          // EVERYWHERE. That is 6,762 of 18,718 genes on CD8 TILs.
          empty: e.p99 <= 0};
}

// output: false because this chunk DEFINES geneNote; without it the function object printed into
// the sidebar as "geneNote = f(g, vals)". The CALL that displays the note is its own chunk below.
// ★ THREE ABSENCE STATES, because "not found" would discard a real finding. The three references
// were built on different feature spaces (19,957 / 27,407 / 70,777 genes against 18,718 shared),
// and which one dropped a gene is a fact about how that reference was constructed.
function geneNote(g, vals) {
  if (!g) return md``;
  if (vals) return md``;
  // ★ TWO SETS, NAMED FOR WHAT THEY ARE. An earlier version built the list of references that DO
  // carry the gene and then wrote a sentence asserting they do NOT, so the page named the carrying
  // reference as the one that was missing it. Keep `carry` and `lack` separate and never derive one
  // sentence from the other list.
  const ME = "cd8_tils";
  const all = ["cd8_tils", "chu_pancancer", "huardb_v2"];
  const carry = all.filter(a => fullVars[a].has(g));
  const lack = all.filter(a => !fullVars[a].has(g));
  const nm = xs => xs.map(a => refName[a]).join(" and ");

  if (carry.includes(ME)) {
    // In this atlas but not shipped, so at least one OTHER reference lacks it.
    const missing = lack;                       // by construction ME is not in here
    return md`**${g} is outside the shared gene universe.** This reference measured it, but
      ${nm(missing)} did not, so it is not shipped here: every gene on these maps is one all three
      atlases tested, the same restriction Figure 4 uses. Comparing on genes only some references
      measured overstates enrichment.`;
  }

  const otherCarriers = carry;                  // ME is absent, so every carrier is another ref
  if (otherCarriers.length === 0)
    return md`**${g} was not measured by any reference in this study.** There is nothing to draw,
      and the absence is not specific to ${refName[ME]}.`;
  if (otherCarriers.length === all.length - 1)
    return md`**${g} was not measured by ${refName[ME]}.** Both other references carry it, so
      the absence really is specific to this one. That is a property of how the reference was
      built, not a gap in the portal.`;
  // Exactly one other carries it, so a third reference lacks it too and the absence is NOT unique.
  return md`**${g} was not measured by ${refName[ME]}.** ${nm(otherCarriers)} carries it and
    ${nm(lack.filter(a => a !== ME))} does not either, so the absence is shared rather than
    specific to this reference.`;
}

ImportantThe two state maps fail in opposite directions

Sampled cells is a stratified sample with a floor per state, so every state stays visible. On the two large atlases it draws only a fraction of the cells; on an atlas shipped whole it is the same cells as the points view, and the provenance note beside this one says which case applies here. All cells (binned) counts every cell, colouring each hexagon by its most common state, with faint bins being mixed ones. But a modal map hides rare states. The sharpest example in this study comes from the huARdb atlas, where Tex-Tpex has 9,496 cells and is the most common state in one hexagon, because it is outvoted almost everywhere it occurs; the same effect applies here. Use the binned view for where the mass is, and the sampled view for where the rare states are. Neither alone is the atlas.
